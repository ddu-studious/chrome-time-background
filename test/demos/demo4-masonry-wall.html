SQL 查询模板
SELECT u.name, COUNT(o.id) AS cnt
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
WHERE u.created_at >= '2026-01-01'
GROUP BY u.id
HAVING cnt > 5
ORDER BY cnt DESC
LIMIT 20;
